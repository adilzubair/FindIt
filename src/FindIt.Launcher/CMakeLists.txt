cmake_minimum_required(VERSION 3.24)
project(FindItLauncher C RC)
add_executable(FindIt WIN32 launcher.c launcher.rc)
target_include_directories(FindIt PRIVATE ../FindIt.App/Assets ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(FindIt PRIVATE advapi32 shell32 user32 gdi32 comctl32)
if(MINGW)
  target_link_options(FindIt PRIVATE -municode -static -Wl,--stack,16777216)
endif()
