cmake_minimum_required(VERSION 3.24)
project(findit_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_WINDOWS_EXPORT_ALL_SYMBOLS ON)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS ON CACHE BOOL "" FORCE)
add_subdirectory(${LLAMA_SOURCE} llama)
add_library(findit_native SHARED embedding.cpp)
target_include_directories(findit_native PRIVATE ${LLAMA_SOURCE}/include ${LLAMA_SOURCE}/tools/mtmd)
target_link_libraries(findit_native PRIVATE llama mtmd)
set_target_properties(findit_native PROPERTIES PREFIX "")
if(USEARCH_SOURCE)
  set(USEARCH_BUILD_LIB_C ON CACHE BOOL "" FORCE)
  set(USEARCH_BUILD_TEST_CPP OFF CACHE BOOL "" FORCE)
  set(USEARCH_BUILD_BENCH_CPP OFF CACHE BOOL "" FORCE)
  set(USEARCH_USE_NUMKONG OFF CACHE BOOL "" FORCE)
  add_subdirectory(${USEARCH_SOURCE} usearch)
endif()
